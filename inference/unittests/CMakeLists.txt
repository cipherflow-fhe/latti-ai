include(CTest)
enable_testing()

add_executable(test_data_structs
  test_data_structs.cpp
  ut_util.cpp
)
target_link_libraries(test_data_structs inference_task)
add_test(NAME test_data_structs COMMAND test_data_structs)

# Copy test_gen_layers.py to build directory
configure_file(
    ${CMAKE_CURRENT_SOURCE_DIR}/test_gen_layers.py
    ${CMAKE_CURRENT_BINARY_DIR}/test_gen_layers.py
    COPYONLY
)

add_executable(test_fhe_layers_hetero test_fhe_layers_hetero.cpp ut_util.cpp)
target_link_libraries(test_fhe_layers_hetero PUBLIC fhe_layers lattisense)

if(INFERENCE_SDK_ENABLE_GPU)
    target_compile_definitions(test_fhe_layers_hetero PRIVATE INFERENCE_SDK_ENABLE_GPU)
endif()

add_test(NAME test_fhe_layers_hetero COMMAND test_fhe_layers_hetero)

add_executable(test_ckks_normalized_l2_direct test_ckks_normalized_l2_direct.cpp)
target_link_libraries(test_ckks_normalized_l2_direct PUBLIC fhe_layers lattisense)
add_test(NAME test_ckks_normalized_l2_direct COMMAND test_ckks_normalized_l2_direct)

# Multi-layer E2E test
# Test data is generated by running test_compiler.py::TestE2EMultipleLayer
add_executable(test_e2e_multiple_layer test_e2e_multiple_layer.cpp)
target_link_libraries(test_e2e_multiple_layer PUBLIC
    -Wl,--start-group
    inference_interface
    inference_task
    fhe_layers
    log
    -Wl,--end-group
    lattisense
    ${HDF5_CXX_LIBRARIES}
)

if(INFERENCE_SDK_ENABLE_GPU)
    target_compile_definitions(test_e2e_multiple_layer PRIVATE INFERENCE_SDK_ENABLE_GPU)
endif()

add_test(NAME test_e2e_multiple_layer COMMAND test_e2e_multiple_layer)
